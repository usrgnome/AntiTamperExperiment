cmake_minimum_required(VERSION 3.20)
project(HardwareBreakpointLab LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use Windows, MSVC and -A x64.")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use a separate build directory.")
endif()
add_executable(hardware_breakpoint_launcher launcher.cpp)
target_compile_features(hardware_breakpoint_launcher PRIVATE cxx_std_17)
target_compile_options(hardware_breakpoint_launcher PRIVATE /W4 /EHsc)
